cmake_minimum_required(VERSION 3.22)
project(DESRUVisuals LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(DF ${CMAKE_CURRENT_SOURCE_DIR}/third_party/doomfix)
file(GLOB_RECURSE DF_SOURCES CONFIGURE_DEPENDS ${DF}/src/common/*.cpp ${DF}/src/features/*.cpp ${DF}/src/layer/*.cpp)
add_library(DESRUVisuals SHARED ${DF_SOURCES} runtime.cpp profile.cpp commands.cpp)
target_include_directories(DESRUVisuals PRIVATE . ${DF}/src third_party/Vulkan-Headers/include)
target_compile_definitions(DESRUVisuals PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE DOOMFIX_LAYER_NAME="DESRUVisuals" DOOMFIX_VERSION="DESRU")
target_compile_options(DESRUVisuals PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(DESRUVisuals PRIVATE user32)
set_target_properties(DESRUVisuals PROPERTIES OUTPUT_NAME "DESRU.Visuals")
set_target_properties(DESRUVisuals PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)

enable_testing()
foreach(test depth render scene solid commands)
  add_executable(cv_${test}_test ${DF}/tests/cv_${test}_test.cpp)
  target_include_directories(cv_${test}_test PRIVATE . ${DF}/src third_party/Vulkan-Headers/include)
  target_compile_definitions(cv_${test}_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
  target_compile_options(cv_${test}_test PRIVATE /utf-8 /EHsc)
  target_link_libraries(cv_${test}_test PRIVATE DESRUVisuals)
  add_test(NAME cv_${test} COMMAND cv_${test}_test)
endforeach()
foreach(test barrier event cv_smoke)
  add_executable(doomfix_${test}_test ${DF}/tests/${test}_test.cpp)
  target_include_directories(doomfix_${test}_test PRIVATE third_party/Vulkan-Headers/include)
  target_compile_definitions(doomfix_${test}_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(doomfix_${test}_test PRIVATE /utf-8 /EHsc)
  target_link_libraries(doomfix_${test}_test PRIVATE user32)
endforeach()
